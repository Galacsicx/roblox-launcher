// Замена для функции proxyImg function proxyImg(url) { if (!url) return ''; // Если картинка уже в формате data: (загружена с ПК), возвращаем как есть if (url.startsWith('data:')) return url; // Пробуем разные прокси для картинок // 1. corsproxy.io - обычно хорошо работает с изображениями // 2. allorigins - резервный вариант // 3. wsrv.nl - если первые два не сработают (маловероятно) // Но так как мы не можем перебирать прокси внутри атрибута src тега img, // мы будем использовать один основной, а если он не сработает, // пользователь увидит onerror и подставится SVG-заглушка. // Поэтому просто ставим corsproxy.io как основной. return 'https://corsproxy.io/?url=' + encodeURIComponent(url); } // Замена для функции fetchRobloxGet async function fetchRobloxGet(url) { // Для API-запросов используем robloxproxy.com, он лучше справляется с API const proxyUrl = 'https://www.robloxproxy.com/' + url.replace('https://', ''); try { const controller = new AbortController(); const timer = setTimeout(() => controller.abort(), 6000); const res = await fetch(proxyUrl, { signal: controller.signal }); clearTimeout(timer); if (!res.ok) { console.warn('Ошибка API прокси:', res.status); return null; } const data = await res.json(); if (data && !data.errors) return data; } catch (e) { console.error('Ошибка запроса к API:', e); } return null; }
